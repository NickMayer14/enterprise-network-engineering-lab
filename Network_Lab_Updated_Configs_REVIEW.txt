===== FORTIGATE-80E =====
config system admin
 edit "admin"
  set password Cisco123
 next
end
config system interface
 edit "port11"
  set vdom "root"
  set ip 10.99.99.1 255.255.255.0
  set allowaccess ping https ssh snmp
  set type physical
  set alias "To_Cisco_Router"
  set role lan
 next
 edit "port12"
  set vdom "root"
  set ip 10.99.100.1 255.255.255.0
  set allowaccess ping https ssh snmp
  set type physical
  set alias "R2_Wan_Backup"
  set role lan
 next
end
config router static
 edit 1
  set dst 10.10.0.0 255.255.0.0
  set gateway 10.99.99.254
  set device "port11"
 next
end
